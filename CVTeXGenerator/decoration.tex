% PNG conversion preserves alpha; PDF uses only a standard grayscale image and alpha mask.
\providecommand{\cvfooterimage}{}
\newcommand{\cvfooterprepared}{}
\directlua{cvfooter = dofile('footer-image.lua')}
\newcommand{\cvrenderfooter}{%
  \directlua{
    local ok, result = pcall(cvfooter.prepare, token.get_macro('cvfooterimage'))
    if ok then token.set_macro('cvfooterprepared', result, 'global')
    else tex.error('Cannot prepare footer PNG', {tostring(result)}) end
  }%
  \includegraphics[width=0.75\textwidth,height=32mm,keepaspectratio]{\cvfooterprepared}%
}
\AddToHook{shipout/background}{%
  \IfFileExists{\cvfooterimage}{%
    \put(\dimexpr0.5\paperwidth\relax,\dimexpr5.5mm-\paperheight\relax){%
      \makebox[0pt][c]{\raisebox{0pt}[0pt][0pt]{\cvrenderfooter}}%
    }%
  }{}%
}
