\newlength{\cvcompanyiconsize}
\setlength{\cvcompanyiconsize}{0.5104167in}
% Each company and role row uses two equal halves of its container.
\newcommand{\cvcompanyicon}[1]{%
  \IfFileExists{#1}{%
    \begin{tikzpicture}
      \path[use as bounding box] (0,0) rectangle (\cvcompanyiconsize,\cvcompanyiconsize);
      \clip[rounded corners=1.5mm] (0.25pt,0.25pt) rectangle
        (\dimexpr\cvcompanyiconsize-0.25pt\relax,\dimexpr\cvcompanyiconsize-0.25pt\relax);
      \node[inner sep=0pt] at (0.5\cvcompanyiconsize,0.5\cvcompanyiconsize) {%
        \includegraphics[width=\cvcompanyiconsize,height=\cvcompanyiconsize,keepaspectratio]{#1}%
      };
    \end{tikzpicture}%
  }{%
    \begin{tikzpicture}
      \path[use as bounding box] (0,0) rectangle (\cvcompanyiconsize,\cvcompanyiconsize);
      \draw[accent!35,line width=0.4pt,rounded corners=1.5mm]
        (0.2pt,0.2pt) rectangle
        (\dimexpr\cvcompanyiconsize-0.2pt\relax,\dimexpr\cvcompanyiconsize-0.2pt\relax);
      \node[text=accent!50,font=\sffamily\scriptsize,inner sep=0pt]
        at (0.5\cvcompanyiconsize,0.5\cvcompanyiconsize) {Icon};
    \end{tikzpicture}%
  }%
}
\newcommand{\cvcompanyidentity}[3]{%
  \begin{minipage}[c]{\cvcompanyiconsize}\cvcompanyicon{#3}\end{minipage}%
  \hspace{3mm}%
  \begin{minipage}[c]{\dimexpr\linewidth-\cvcompanyiconsize-3mm\relax}\raggedright
    {\sffamily\bfseries\color{accent}#1}\par
    #2\par
  \end{minipage}%
}

% Institution details and a compact right-aligned period/status column.
\newcommand{\cveducation}[5]{%
  \par\addvspace{7pt}%
  \settowidth{\cvperiodwidth}{{\small #3}}%
  \settowidth{\cvcontractwidth}{{\small\textit{(#4)}}}%
  \ifdim\cvcontractwidth>\cvperiodwidth
    \setlength{\cvperiodwidth}{\cvcontractwidth}%
  \fi
  \noindent\begin{minipage}[c]{\dimexpr\linewidth-\cvperiodwidth-3mm\relax}%
    \cvcompanyidentity{#1}{#2}{#5}%
  \end{minipage}\hfill
  \begin{minipage}[c]{\cvperiodwidth}\centering\small
    \mbox{#3}\par
    \if\relax\detokenize{#4}\relax\else\textit{(#4)}\par\fi
  \end{minipage}\par
}
